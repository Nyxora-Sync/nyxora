## Learn directly on the site

Nyxora Sync is built for people who want to learn by reading, practicing, and building — not by assembling a local development environment.

Everything you need is on the live site:

- **194 unique written lessons** across web development, specializations, and programming language paths
- **Free, curated resources** in every lesson, with optional "go deeper" links
- **Interactive playgrounds** for HTML, CSS, JavaScript, React, Node, and Express
- **Progress tracking** with XP, levels, streaks, badges, notes, and bookmarks
- **Verifiable track certificates** with public verification links and print-ready pages
- **Lesson discussions**, a project showcase, public learner profiles, and moderation tools

No paywalls. No ads. No account required to start.

## Deployment

Production: https://nyxora-sync.tech

The public URL is served by the custom domain `nyxora-sync.tech` pointing directly to Vercel (bypassing the Cloudflare Worker proxy for POST support). The Next.js app runs on **Vercel** in the `iad1` region, co-located with the Neon database in `us-east-1`.

### App (Vercel)

```bash
npx vercel login
npx vercel link --scope miyazaki-akis-projects --project nyxora-sync
npm run deploy:web
```

The project is `miyazaki-akis-projects/nyxora-sync`. Production builds apply pending Prisma migrations via the Vercel build command, guarded by `VERCEL_ENV=production`; preview builds run `npm run build` without touching the database. The old `postbuild` `prisma db push --accept-data-loss` step has been removed.

Set these environment variables for Production:

- `DATABASE_URL` and `DIRECT_URL` (Neon)
- `AUTH_SECRET`, `AUTH_TRUST_HOST=true`
- `AUTH_URL=https://nyxora-sync.tech`
- `NEXTAUTH_URL=https://nyxora-sync.tech`
- `NEXT_PUBLIC_SITE_URL=https://nyxora-sync.tech`
- `GITHUB_CLIENT_ID`, `GITHUB_CLIENT_SECRET`
- `GOOGLE_CLIENT_ID`, `GOOGLE_CLIENT_SECRET`
- `ADMIN_EMAILS` (comma-separated maintainer emails; gates `/admin/messages`, `/admin/newsletter`, and `/admin/reports`)
- `BLOB_READ_WRITE_TOKEN` (enables project screenshot uploads through Vercel Blob; the `nyxora-uploads` store is connected to Production and Preview. Pull it with `vercel env pull` when testing uploads locally.)

`AUTH_URL`, `NEXTAUTH_URL`, and `NEXT_PUBLIC_SITE_URL` must all be the public origin (`https://nyxora-sync.tech`). The apex domain is the canonical host; `www.nyxora-sync.tech` permanently redirects to it. Vercel is configured for the `iad1` region so functions run next to the Neon database. Disable **Deployment Protection → Vercel Authentication** for production. Connecting the GitHub repository in the Vercel dashboard enables automatic deploys; otherwise run `npm run deploy:web`.

### Health and monitoring

`GET /api/health` checks database connectivity and returns `200` with `{ status: "ok" }` or `503` when degraded. Point an uptime monitor at it. Vercel Speed Insights and function logs cover performance and server errors; set up a Sentry DSN only if you need long-retention client error traces.

import { readdirSync, readFileSync, writeFileSync, mkdirSync } from "node:fs";
import { join, relative, sep } from "node:path";

const SKIP_DIRS = new Set([
  ".git", ".next", ".vercel", ".wrangler", "node_modules",
  "test-results", "migrations", "bundles",
]);
const SKIP_PATHS = ["src/content"]; // lesson .mdx files; send a sample separately if needed
const SKIP_FILES = /^(package-lock\.json|tsconfig\.tsbuildinfo|next-env\.d\.ts|LICENSE.*|favicon\.ico)$/;
const TEXT_EXT = /\.(ts|tsx|mjs|css|json|jsonc|md|yml|prisma|toml|webmanifest|js)$/;
const MAX_CHARS = 400_000; // split a bundle into parts above this size

function walk(dir, out = []) {
  for (const entry of readdirSync(dir, { withFileTypes: true })) {
    const full = join(dir, entry.name);
    const rel = relative(".", full).split(sep).join("/");
    if (entry.isDirectory()) {
      if (SKIP_DIRS.has(entry.name) || SKIP_PATHS.includes(rel)) continue;
      walk(full, out);
    } else {
      if (SKIP_FILES.test(entry.name)) continue;
      if (entry.name.startsWith(".env") && entry.name !== ".env.example") continue;
      if (!TEXT_EXT.test(entry.name) && entry.name !== ".env.example") continue;
      out.push(rel);
    }
  }
  return out;
}

function groupOf(p) {
  if (p.startsWith("src/components/")) return "components";
  if (p.startsWith("src/app/api/")) return "api-routes";
  if (p.startsWith("src/app/")) return "app-pages";
  if (/^src\/(lib|hooks|middleware|types)\//.test(p) || p === "src/mdx-components.tsx") return "logic";
  if (/^(prisma|e2e|scripts|proxy-worker)\//.test(p)) return "prisma-tests-infra";
  return "config-and-docs";
}

const groups = {};
for (const p of walk(".").sort()) (groups[groupOf(p)] ??= []).push(p);

mkdirSync("bundles", { recursive: true });
for (const [name, files] of Object.entries(groups)) {
  let part = 1, buf = "";
  const flush = () => {
    if (!buf) return;
    writeFileSync(`bundles/${name}-${part++}.txt`, buf);
    buf = "";
  };
  for (const f of files) {
    const chunk = `\n===== FILE: ${f} =====\n${readFileSync(f, "utf8")}\n`;
    if (buf.length + chunk.length > MAX_CHARS) flush();
    buf += chunk;
  }
  flush();
}
console.log("Done. Upload everything in the ./bundles folder.");

# Community Features

Community features turn Nyxora Sync from a course into a place to learn with other people.
Everything here is optional: without `DATABASE_URL` the lessons, playground, and local progress
tracking keep working, and the community routes return a clear `503` instead of crashing.

## What exists

| Surface | Route | Source |
| --- | --- | --- |
| Community hub | `/community` | `src/app/community/page.tsx` |
| Project showcase | `/community/showcase` | `src/app/community/showcase/page.tsx` |
| Project detail + discussion | `/community/showcase/[id]` | `src/app/community/showcase/[id]/page.tsx` |
| Public learner profile | `/community/people/[id]` | `src/app/community/people/[id]/page.tsx` |
| Profile & privacy settings | `/settings` | `src/app/settings/page.tsx` |
| Lesson discussion | bottom of every lesson | `src/components/lesson-client.tsx` |
| Contribution guide | `/contribute` | `src/app/contribute/page.tsx` |

`/showcase` permanently redirects to `/community/showcase` for old links.

## Data model

Added in `prisma/migrations/20261019213529_community`:

- `User.bio`, `website`, `githubUrl`, `twitterUrl` — optional public profile fields.
- `Project` — existing showcase model. `upvotes` is a denormalized counter used for sorting.
- `ProjectUpvote` — one row per `(projectId, userId)`. The counter and the row move in the same
  transaction; the unique constraint makes double-clicks harmless.
- `Comment` — polymorphic by `targetType` (`lesson` | `project`) + `targetId`. Replies are one
  level deep (`parentId` points at a top-level comment). Deleting a parent cascades to replies.

Target ids are validated on write: lesson ids must exist in `src/lib/curriculum.ts`, project ids
must exist in the database. This keeps orphaned discussions out of the table.

## API

All routes live under `src/app/api/community/`. Auth is the NextAuth session (`auth()`); the
database is Prisma.

| Method | Path | Auth | Behavior |
| --- | --- | --- | --- |
| `GET` | `/api/community/projects` | public | Cursor pagination; `track`, `q`, `sort=newest\|popular`, `limit` (max 48). Includes comment counts and viewer upvote state. |
| `POST` | `/api/community/projects` | required | Zod validated; max 5 projects per user per 24h. |
| `GET` | `/api/community/projects/[id]` | public | Detail + `isOwner` + viewer upvote state. |
| `PATCH` | `/api/community/projects/[id]` | owner | Partial update; returns the full updated project detail so the UI can refresh in place. |
| `DELETE` | `/api/community/projects/[id]` | owner | Deletes project comments, then the project (upvotes cascade). |
| `POST` | `/api/community/projects/[id]/upvote` | required | Toggle; returns `{ upvoted, upvotes }`. |
| `GET` | `/api/community/comments` | public | `targetType` + `targetId`; newest 200 top-level comments with up to 50 replies each, nested. |
| `POST` | `/api/community/comments` | required | One-level replies; max 12 comments per user per 10 minutes. |
| `DELETE` | `/api/community/comments/[id]` | author | Replies cascade. |
| `GET` | `/api/community/profile` | required | Own profile + preferences (includes email; never public). |
| `PATCH` | `/api/community/profile` | required | Profile fields and `showProgress` / `emailUpdates`. |
| `GET` | `/api/community/people/[id]` | public | Public profile. Progress is hidden unless `showProgress` is true or the viewer is the owner. |
| `GET` | `/api/community/stats` | public | Real counts used on the community hub — no invented numbers. |

Every route starts with `communityEnabled()` and returns
`503 COMMUNITY_UNAVAILABLE` when `DATABASE_URL` is missing.

## Rules and safety

- Ownership: only a project owner can edit or delete it; only a comment author can delete it.
  Owners can edit every project field, including the screenshot URL, without deleting and
  republishing it. There is no admin/moderator role yet.
- Privacy: email is never returned by public routes. `showProgress = false` hides XP, completed
  lessons, streaks, and badges from other learners.
- URLs: project and profile links must be valid `http(s)` URLs. User text is rendered as text
  (React escapes it), and newlines are preserved with `whitespace-pre-wrap`.
- Rate limits use row counts over a sliding window, so they need no extra infrastructure.
- Profiles with no name, bio, or image are not published (the owner can still open their own).

## UI components

- `src/components/community/showcase-client.tsx` — search, track filter, sort, pagination,
  empty/loading/error states.
- `src/components/community/project-card.tsx`, `upvote-button.tsx`, `project-form-dialog.tsx`. The dialog handles both publishing and editing; owners edit from the project detail page.
- `src/components/community/comment-thread.tsx` — shared by lessons and project pages.
- `src/components/community/profile-client.tsx`, `community-stats.tsx`.
- `src/components/settings/settings-client.tsx`.

`CommunityStats` fetches real aggregate counts after mount; the server-rendered shell shows
dashes, so the page stays fast and degrades cleanly if the API is unavailable.

# Writing lessons

Lessons are written **on the site**, not outsourced to links. Each lesson lives in its own MDX file:

```
src/content/lessons/<lesson-id>.mdx
```

The lesson ID comes from `src/lib/curriculum.ts`. The external resources attached to a lesson are **supplementary** — the MDX file is the primary material.

## Required structure

1. **Opening hook** (2 short paragraphs): why this topic matters and what the reader will be able to do by the end.
2. **3–7 `##` sections**, each teaching one idea. Use `###` only for sub-points.
3. **At least two code examples** in fenced blocks with a language tag.
4. **At least one `<Callout>`** for a tip, warning, or common mistake.
5. **At least one `<TryIt>`** with a small runnable example.
6. **1–2 `<Quiz>` questions** near the end.
7. **One `<Exercise>`** with concrete, checkable tasks.
8. **`## Summary`** with 4–6 bullets, then one sentence bridging to the next lesson.

Target **1,000–1,800 words**. Depth over padding: every paragraph should teach, warn, or demonstrate something.

## Components

### Callout

```mdx
<Callout type="tip" title="Optional title">
  One or two sentences. Types: `info`, `tip`, `warning`, `danger`, `success`.
</Callout>
```

### Key idea

```mdx
<KeyPoint>
  The one sentence a reader should remember if they forget everything else.
</KeyPoint>
```

### Try it

````mdx
<TryIt title="Try it yourself">
```html
<h1>Hello</h1>
```
</TryIt>
````

### Exercise

```mdx
<Exercise title="Build a profile card" difficulty="medium">
  1. First task
  2. Second task
</Exercise>
```

Difficulty is `easy`, `medium`, or `hard`.

### Quiz

```mdx
<Quiz
  question="What does HTTP stand for?"
```

`answer` is the **zero-based index** of the correct option. Write plausible distractors, not jokes.

## Voice and style

- Direct, second person: “You send a request”, not “the developer sends a request”.
- Short paragraphs. One idea per paragraph.
- Define a term the first time it appears, in plain language.
- Prefer concrete examples over abstractions: a real URL, a real JSON body, a real error.
- No emojis. No hype (“simply”, “just”, “obviously”).
- Explain *why*, not only *what*.
- When showing a command, say where to run it and what the expected output looks like.
- End sections with the consequence: what breaks if you get this wrong?

## Accuracy rules

- Target **Node.js 20+**, **Express 4** (code should also run on Express 5), **Mongoose 8**, **React 19**, and **React Router v6/v7**.
- Use `async/await` for asynchronous code.
- Use ESM (`import` / `export`) in new examples, and show `"type": "module"` in `package.json` when it matters.
- Never commit real secrets. Use `.env` and `process.env`.
- Do not teach deprecated APIs (`componentWillMount`, `findByIdAndRemove`, callback-style Mongoose, etc.).
- Validate code by running it or by checking the API docs. If something is version-sensitive, say so.
- Prefer web-standard APIs over library-specific shortcuts.

## MDX gotchas

- **Do not start with an H1.** The page renders the lesson title from the curriculum. Begin with prose, then `##`.
- Escape `<` and `{` in prose. For HTML tags, use inline code: `` `<div>` ``.
- Every table needs a header row and separator.
- Fenced code blocks need a language: `js`, `ts`, `jsx`, `html`, `css`, `bash`, `http`, `json`, `text`.
- No frontmatter. Metadata lives in `src/lib/curriculum.ts`.
- JSX components must be separated from Markdown blocks by blank lines.
- External links are normal Markdown; they open in a new tab automatically.
- Images use normal Markdown syntax; always include alt text. Prefer a `text` code-fence diagram when the point is structural.

# Newsletter and contact operations

> **Newsletter status: disabled.** The newsletter is switched off with
> `NEWSLETTER_ENABLED = false` in `src/lib/newsletter-feature.ts`. No signup form
> or navigation link is rendered, `/newsletter` and its confirmation and
> unsubscribe pages redirect home, and the newsletter and admin API routes return
> `404`. The implementation, database models, and existing addresses are
> preserved. See [Re-enabling the newsletter](#re-enabling-the-newsletter) before
> turning it back on.
>
> The contact form is still active. It stores data in the shared PostgreSQL
> database and degrades to a `503` when `DATABASE_URL` is missing.

## Newsletter

The newsletter uses double opt-in. A signup is **not** added to the sending list
until the address opens the confirmation email and presses the confirm button.

### Signup and confirmation flow

1. The footer or `/newsletter` form posts to `POST /api/newsletter`.
2. The route validates the email, applies the global rate limit, and upserts one
   `NewsletterSubscriber` row per address.
3. If the address is unconfirmed, the route sends one confirmation email. The
   signed link is valid for 48 hours.
4. `POST /api/newsletter/confirm` verifies the signed token and sets
   `confirmedAt`. The page at `/newsletter/confirm?token=...` performs the call
   from a button; the page itself does not mutate anything.
5. Only rows with `confirmedAt` set are included in a broadcast.

If the address is already confirmed, submitting the form again returns
`alreadyConfirmed` and sends nothing. If a confirmation was sent in the last 60
seconds, the route returns `alreadySent` instead of sending a duplicate.

### Signed links

Confirmation and unsubscribe tokens are HMAC-SHA256 signed, contain the
subscriber id, purpose, and expiry, and are verified without a token table.
`NEWSLETTER_SECRET` is optional; the code falls back to `AUTH_SECRET`.
Rotating either secret invalidates links already in inboxes.

- Confirmation links expire after 48 hours.
- Unsubscribe links expire after 10 years, so old newsletters still work.
- Unsubscribe links appear in every broadcast. Unsubscribing deletes the row
  immediately instead of adding a suppression record.

### API

| Method | Path | Auth | Behavior |
| --- | --- | --- | --- |
| `POST` | `/api/newsletter` | public | `{ email, source, website }`; `source` is `footer`, `newsletter`, or `site`; `website` is a honeypot. Returns `confirmationRequired`, `alreadyConfirmed`, or `alreadySent`. |
| `DELETE` | `/api/newsletter` | public | `{ email }` or `{ token }`; deletes the address and always succeeds for unknown email addresses. |
| `POST` | `/api/newsletter/confirm` | public | `{ token }`; idempotently marks the subscription confirmed. |
| `POST` | `/api/newsletter/unsubscribe` | public | `{ token }` or `?token=...`; used by the unsubscribe page and by mailbox-provider one-click unsubscribe. |
| `POST` | `/api/admin/newsletter` | admin | Sends a test or broadcast. See below. |

Broadcasts include:

```text
List-Unsubscribe: <https://your-site/api/newsletter/unsubscribe?token=...>
List-Unsubscribe-Post: List-Unsubscribe=One-Click
```

Mailbox providers can therefore POST to the API with no JSON body. The token is
read from the query string.

### Admin broadcasts

Open `/admin/newsletter` while signed in with an address listed in
`ADMIN_EMAILS` (comma separated, case-insensitive). Leave `ADMIN_EMAILS` empty
to disable the page. The page is `noindex` and is not linked from public
navigation.

The composer is plain text only, which avoids HTML injection and keeps the
emails readable in every client. It supports:

- **Send test to me** — prefixes the subject with `[TEST]` and uses a placeholder
  unsubscribe link. Test sends are recorded in the audit table.
- **Send to confirmed subscribers** — requires typing `SEND`. There is no undo,
  scheduling, or automatic retry.

Limits and safety:

- 500 confirmed recipients per request. For a larger list, send in batches or
  replace this with a queue before the list grows.
- One real broadcast per minute, to catch accidental double clicks.
- Partial failures return the sent/failed counts and are recorded; the admin
  must inspect the mail server before retrying.
- The message body is not stored. `NewsletterSend` records the sender, subject,
  recipient count, failure count, test flag, and date.

### Re-enabling the newsletter

1. Set `NEWSLETTER_ENABLED` to `true` in `src/lib/newsletter-feature.ts`.
2. Add working SMTP credentials for the production environment. These normally
   come from the email provider's dashboard; do not copy placeholder values from
   `.env.example`.
3. Set `ADMIN_EMAILS` to the exact email address used to sign in to the admin
   account. Optionally set `NEWSLETTER_SECRET`; otherwise the code falls back to
   `AUTH_SECRET`.
4. Confirm `NEXT_PUBLIC_SITE_URL` is the public production origin, or let Vercel
   derive it from `VERCEL_PROJECT_PRODUCTION_URL`, so confirmation and
   unsubscribe links point at the live domain.
5. Redeploy. Environment variable changes do not apply to an existing build.
6. Submit a real address, open the confirmation email, and complete a test send
   before broadcasting to the list.

Existing rows from before double opt-in are unconfirmed and will not receive
broadcasts. Any address submitted before the newsletter was disabled must be
submitted again to receive a fresh confirmation email.

### Email configuration

Newsletter signup and broadcasts require SMTP. The code accepts either:

```bash
EMAIL_SERVER_HOST=...
EMAIL_SERVER_PORT=587
EMAIL_SERVER_USER=...
EMAIL_SERVER_PASSWORD=...
EMAIL_FROM="Nyxora Sync <newsletter@your-domain>"
```

or the legacy URL form:

```bash
EMAIL_SERVER="smtp://user:pass@host:587"
EMAIL_FROM="Nyxora Sync <newsletter@your-domain>"
```

`EMAIL_FROM` is required. Without usable SMTP, signup returns a friendly `503`
instead of silently storing an address that can never be confirmed.

### Production checklist

In Vercel, open **Project → Settings → Environment Variables → Production** and
add the SMTP values supplied by your email provider:

- `EMAIL_SERVER_HOST`
- `EMAIL_SERVER_PORT`
- `EMAIL_SERVER_USER`
- `EMAIL_SERVER_PASSWORD`
- `EMAIL_FROM`

Then add `ADMIN_EMAILS` with the exact email address you use to sign in to the
site. Optionally add `NEWSLETTER_SECRET`; otherwise the site falls back to
`AUTH_SECRET`. Confirm that `NEXT_PUBLIC_SITE_URL` is the public production
origin (or let Vercel derive it from `VERCEL_PROJECT_PRODUCTION_URL`) so
confirmation and unsubscribe links point at the live domain.

Redeploy after changing environment variables. While the feature flag is off,
the API returns `404` and no address is stored. When the flag is on but SMTP is
not configured, `POST /api/newsletter` returns `503` and no address is stored.

### Database and operations

The double opt-in migration added:

- `NewsletterSubscriber.confirmedAt` — null until the email is confirmed.
- `NewsletterSubscriber.confirmationSentAt` — used for the 60-second resend
  throttle.
- `NewsletterSubscriber.updatedAt` — row update timestamp.
- `NewsletterSend` — metadata-only audit trail for test and real sends.

Existing rows from before double opt-in are unconfirmed. They do not receive
broadcasts. To let one of those addresses join, submit it again through the
signup form so it gets a fresh confirmation email.

Read or export the list with:

```bash
npm run db:studio
```

Open `NewsletterSubscriber`. `confirmedAt IS NULL` means the address never
completed confirmation. `NewsletterSend` contains the send history. The global
limit is 200 signup requests per hour; it is a blunt abuse guard, not an
analytics metric.

### Not built

- No queue, scheduler, retry worker, or bounce/complaint handling.
- No open, click, or conversion tracking.
- No HTML composer or per-segment audiences.
- No suppression list; unsubscribe deletes the row.

For a much larger list, move sends to a background job and use a dedicated
email provider with bounce webhooks.

## Contact

- Route: `POST /api/contact` with `{ name, email, topic, message, website }`.
- Topics: `general`, `bug`, `content`, `conduct`, `privacy`, `security`.
- The message is validated (20–4000 characters), stored in `ContactMessage`, and then
  a notification email is attempted.
- `CONTACT_INBOX` is the notification recipient. It falls back to `EMAIL_FROM`.
- A mail failure is logged but does not fail the request: the message is already stored.
- Limits: three messages per sender email per hour and 100 messages per hour globally.
- The same `website` honeypot is used.

Read messages with `npm run db:studio` and open `ContactMessage`. Filter on
`handled = false`, and mark rows as handled after replying. Contact messages are not
visible in any public UI.

## Privacy notes

- Newsletter addresses are used only for curriculum updates. They are deleted on
  unsubscribe, and only confirmed addresses receive broadcasts.
- Contact messages are private to the maintainers. Do not paste them into public
  issues without the sender's permission.
- Both tables are covered by the site's privacy policy. If the retention or purpose
  changes, update that page and the date at the top.

# SEO and metadata

Most SEO behavior is centralized in `src/lib/seo.ts`.

## Canonical origin

The public origin is resolved in this order:

1. `NEXT_PUBLIC_SITE_URL`
2. `VERCEL_PROJECT_PRODUCTION_URL` (Vercel's stable production hostname)
3. `http://localhost:3000`

Set `NEXT_PUBLIC_SITE_URL` to the final public origin when a custom domain is
added. It must not have a trailing slash. Canonical tags, Open Graph URLs, the
sitemap, and JSON-LD all use this value.

## Page metadata

Public pages call `pageMetadata()` with a title, description, and path. It adds:

- a self-referencing canonical URL
- Open Graph and Twitter card fields
- the shared 1200×630 image at `/og-image.jpg`
- optional `noindex` for private pages

Private or thin pages (`/dashboard`, `/settings`, `/auth/*`, public profile and
project detail shells) are `noindex`. They are also excluded from the sitemap and
most are disallowed in `robots.txt`.

## Shared module canonicalization

The beginner and developer tracks share four core modules: `node-express`,
`mongodb`, `react`, and `fullstack`. Those pages render the same lessons at two
URLs. `canonicalLearnPath()` points both copies at the developer-track URL, and
the sitemap only lists the canonical copy.

Unique beginner modules such as `foundations` remain canonical on the beginner
track.

## Sitemap and robots

- `/sitemap.xml` is generated from `src/app/sitemap.ts`.
- `/robots.txt` is generated from `src/app/robots.ts`.
- Both are static routes and are regenerated on deploy.
- The sitemap contains canonical public URLs: static pages, all three tracks,
  and every unique module and lesson. `/newsletter` is included only when the
  newsletter feature flag is enabled.

## Structured data

`src/components/structured-data.tsx` renders escaped JSON-LD.

- Every page: `Organization` and `WebSite`.
- Track pages: `Course` and `BreadcrumbList`.
- Module pages: `BreadcrumbList`.
- Lesson pages: `LearningResource` and `BreadcrumbList`.

Validate changes with the [Schema Markup Validator](https://validator.schema.org/)
and Google's [Rich Results Test](https://search.google.com/test/rich-results).

## After a domain change

1. Set `NEXT_PUBLIC_SITE_URL` in the hosting environment.
2. Redeploy.
3. Verify canonical tags, `robots.txt`, and `sitemap.xml` use the new origin.
4. Add the domain to Google Search Console and submit `/sitemap.xml`.

import type { Instrumentation } from 'next'

/**
 * Structured server-error logging. Vercel captures stdout/stderr, so this is
 * enough to search failures by route without shipping a third-party SDK.
 * Swap the console call for Sentry/Datadog here if long retention is needed.
 */
export const onRequestError: Instrumentation.onRequestError = async (
  error,
  request,
  context
) => {
  const message = error instanceof Error ? error.message : String(error)
  const digest =
    typeof error === 'object' && error !== null && 'digest' in error
      ? String(error.digest)
      : undefined

  console.error(
    JSON.stringify({
      type: 'request_error',
      message,
      digest,
      path: request.path,
      method: request.method,
      routerKind: context.routerKind,
      routePath: context.routePath,
      routeType: context.routeType,
      renderSource: context.renderSource,
      timestamp: new Date().toISOString(),
    })
  )
}

/* Nyxora Sync app-shell service worker.
 * Deliberately conservative: only static assets are cached. Navigations are
 * network-first and fall back to /offline, so one learner's HTML is never
 * served to another. API and auth requests always go to the network.
 */
const VERSION = 'nyxora-v1'
const OFFLINE_CACHE = `${VERSION}-offline`
const STATIC_CACHE = `${VERSION}-static`
const OFFLINE_URL = '/offline'

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches
      .open(OFFLINE_CACHE)
      .then((cache) =>
        cache.addAll([
          OFFLINE_URL,
          '/site-mark.webp',
          '/icon-192.png',
          '/icon-512.png',
        ])
      )
      .then(() => self.skipWaiting())
  )
})

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches
      .keys()
      .then((keys) =>
        Promise.all(
          keys
            .filter((key) => !key.startsWith(VERSION))
            .map((key) => caches.delete(key))
        )
      )
      .then(() => self.clients.claim())
  )
})

function isStaticAsset(url) {
  return (
    url.pathname.startsWith('/_next/static/') ||
    /\.(?:css|js|woff2?|png|jpe?g|webp|avif|gif|svg|ico)$/i.test(url.pathname)
  )
}

self.addEventListener('fetch', (event) => {
  const request = event.request
  if (request.method !== 'GET') return

  const url = new URL(request.url)
  if (url.origin !== self.location.origin) return
  if (
    url.pathname.startsWith('/api/') ||
    url.pathname.startsWith('/auth/') ||
    url.pathname.startsWith('/admin/')
  ) {
    return
  }

  if (request.mode === 'navigate') {
    event.respondWith(
      fetch(request).catch(async () => {
        const cached = await caches.match(OFFLINE_URL)
        return (
          cached ||
          new Response('You are offline. Reconnect and try again.', {
            status: 503,
            headers: { 'Content-Type': 'text/plain; charset=utf-8' },
          })
        )
      })
    )
    return
  }

  if (!isStaticAsset(url)) return

  event.respondWith(
    caches.open(STATIC_CACHE).then(async (cache) => {
      const cached = await cache.match(request)
      if (cached) return cached

      const response = await fetch(request)
      if (response.ok) {
        cache.put(request, response.clone()).catch(() => {})
      }
      return response
    })
  )
})
