import { auth } from '@/lib/auth/auth'
import { isAdmin } from '@/lib/admin'
import { jsonError } from '@/lib/community'
import { prisma } from '@/lib/db'

type RouteContext = { params: Promise<{ id: string }> }

export async function PATCH(request: Request, { params }: RouteContext) {
  const session = await auth()
  if (!isAdmin(session?.user?.email)) {
    return jsonError(404, 'NOT_FOUND', 'Message not found.')
  }

  if (!process.env.DATABASE_URL) {
    return jsonError(503, 'CONTACT_UNAVAILABLE', 'The contact inbox is temporarily unavailable.')
  }

  const body = (await request.json().catch(() => null)) as { handled?: unknown } | null
  if (!body || typeof body.handled !== 'boolean') {
    return jsonError(422, 'VALIDATION_ERROR', 'Choose whether the message is handled.')
  }

  const { id } = await params
  const updated = await prisma.contactMessage.updateMany({
    where: { id },
    data: { handled: body.handled },
  })

  if (updated.count === 0) {
    return jsonError(404, 'NOT_FOUND', 'Message not found.')
  }

  return Response.json(
    { data: { id, handled: body.handled } },
    { headers: { 'Cache-Control': 'no-store' } }
  )
}

import { z } from 'zod'
import { auth } from '@/lib/auth/auth'
import { prisma } from '@/lib/db'
import { jsonError } from '@/lib/community'
import {
  isNewsletterAdmin,
  newsletterEmailConfigured,
  sendNewsletterBroadcast,
} from '@/lib/newsletter'
import { siteUrl } from '@/lib/seo'
import { NEWSLETTER_ENABLED } from '@/lib/newsletter-feature'

// Sending to a small list can take longer than the default route budget. The
// admin UI is intentionally limited to a few hundred recipients per send.
export const maxDuration = 60

const MAX_RECIPIENTS = 500

const schema = z
  .object({
    subject: z
      .string()
      .trim()
      .min(3, 'Write a subject')
      .max(140, 'Keep the subject under 140 characters')
      .refine((value) => !/[\r\n]/.test(value), 'The subject must be a single line'),
    body: z
      .string()
      .trim()
      .min(20, 'Write at least a couple of sentences')
      .max(20_000, 'Keep the email under 20,000 characters'),
    test: z.boolean().optional().default(false),
    confirm: z.string().optional(),
  })
  .strict()

export async function POST(request: Request) {
  if (!NEWSLETTER_ENABLED) {
    return jsonError(404, 'NEWSLETTER_DISABLED', 'The newsletter is not available.')
  }

  if (!process.env.DATABASE_URL) {
    return jsonError(503, 'NEWSLETTER_UNAVAILABLE', 'The newsletter database is not configured.')
  }

  const session = await auth()
  const adminEmail = session?.user?.email
  if (!isNewsletterAdmin(adminEmail)) {
    return jsonError(403, 'FORBIDDEN', 'You do not have access to the newsletter admin.')
  }
  if (!newsletterEmailConfigured()) {
    return jsonError(503, 'EMAIL_UNAVAILABLE', 'Email delivery is not configured.')
  }

  const body = await request.json().catch(() => null)
  const parsed = schema.safeParse(body)
  if (!parsed.success) {
    return Response.json(
      {
        error: {
          code: 'VALIDATION_ERROR',
          message: parsed.error.issues[0]?.message ?? 'Check the subject and body.',
        },
      },
      { status: 422 }
    )
  }

  const { subject, body: text, test, confirm } = parsed.data

  try {
    if (test) {
      const result = await sendNewsletterBroadcast({
        subject,
        body: text,
        recipients: [],
        origin: siteUrl,
        testTo: adminEmail!,
      })

      await prisma.newsletterSend.create({
        data: {
          sentBy: adminEmail!,
          subject,
          recipientCount: 1,
          failedCount: 0,
          isTest: true,
        },
      })

      return Response.json({ data: { ...result, total: 1, test: true } })
    }

    if (confirm !== 'SEND') {
      return jsonError(400, 'CONFIRMATION_REQUIRED', 'Type SEND to confirm a real broadcast.')
    }

    const lastSend = await prisma.newsletterSend.findFirst({
      where: { isTest: false },
      orderBy: { createdAt: 'desc' },
      select: { createdAt: true },
    })
    if (lastSend && Date.now() - lastSend.createdAt.getTime() < 60_000) {
      return jsonError(
        429,
        'RATE_LIMITED',
        'A broadcast was just sent. Wait a minute before sending another one.'
      )
    }

    const recipients = await prisma.newsletterSubscriber.findMany({
      where: { confirmedAt: { not: null } },
      select: { id: true, email: true },
      orderBy: { createdAt: 'asc' },
      take: MAX_RECIPIENTS + 1,
    })

    if (recipients.length === 0) {
      return jsonError(400, 'NO_RECIPIENTS', 'There are no confirmed subscribers yet.')
    }
    if (recipients.length > MAX_RECIPIENTS) {
      return jsonError(
        400,
        'TOO_MANY_RECIPIENTS',
        `This broadcast has more than ${MAX_RECIPIENTS} recipients. Send it in smaller batches.`
      )
    }

    const result = await sendNewsletterBroadcast({
      subject,
      body: text,
      recipients,
      origin: siteUrl,
    })

    await prisma.newsletterSend.create({
      data: {
        sentBy: adminEmail!,
        subject,
        recipientCount: recipients.length,
        failedCount: result.failed,
        isTest: false,
      },
    })

    if (result.sent === 0) {
      return Response.json(
        {
          error: {
            code: 'SEND_FAILED',
            message: 'Every recipient failed. Check the email server configuration and try again.',
          },
          data: { ...result, total: recipients.length },
        },
        { status: 502 }
      )
    }

    return Response.json({
      data: { ...result, total: recipients.length, test: false },
    })
  } catch (error) {
    console.error(
      'Newsletter broadcast failed:',
      error instanceof Error ? error.message : error
    )
    return jsonError(
      502,
      'SEND_FAILED',
      'The email server did not accept the send. Check the server logs and try again.'
    )
  }
}

import { z } from 'zod'
import { auth } from '@/lib/auth/auth'
import { isAdmin } from '@/lib/admin'
import { prisma } from '@/lib/db'
import { jsonError } from '@/lib/community'
import { getRequestIp, hashIp, writeAuditLog } from '@/lib/security'

const actionSchema = z
  .object({
    action: z.enum(['resolve', 'dismiss', 'delete']),
    note: z.string().trim().max(1000).optional(),
  })
  .strict()

type RouteContext = { params: Promise<{ id: string }> }

export async function PATCH(request: Request, { params }: RouteContext) {
  const session = await auth()
  const adminId = session?.user?.id
  if (!adminId || !isAdmin(session?.user?.email)) {
    return jsonError(403, 'FORBIDDEN', 'Maintainer access required.')
  }

  const { id } = await params
  const body = await request.json().catch(() => null)
  const parsed = actionSchema.safeParse(body)
  if (!parsed.success) {
    return jsonError(422, 'VALIDATION_ERROR', 'Choose resolve, dismiss, or delete.')
  }

  const report = await prisma.report.findUnique({ where: { id } })
  if (!report) return jsonError(404, 'NOT_FOUND', 'Report not found.')

  const { action, note } = parsed.data
  const ipHash = hashIp(getRequestIp(request))
  const reviewedAt = new Date()

  if (action === 'delete') {
    if (report.targetType === 'profile') {
      return jsonError(400, 'UNSUPPORTED', 'Profile reports cannot delete an account from this queue.')
    }

    if (report.targetType === 'project') {
      await prisma.$transaction([
        prisma.comment.deleteMany({
          where: { targetType: 'project', targetId: report.targetId },
        }),
        prisma.project.deleteMany({ where: { id: report.targetId } }),
      ])
    } else {
      const comment = await prisma.comment.findUnique({
        where: { id: report.targetId },
        select: { parentId: true },
      })
      if (comment) {
        await prisma.comment.deleteMany({
          where: comment.parentId
            ? { id: report.targetId }
            : { OR: [{ id: report.targetId }, { parentId: report.targetId }] },
        })
      }
    }

    await prisma.report.updateMany({
      where: {
        targetType: report.targetType,
        targetId: report.targetId,
        status: 'open',
      },
      data: {
        status: 'resolved',
        resolutionNote: note || 'Content removed by a maintainer.',
        reviewedById: adminId,
        reviewedAt,
      },
    })

    await writeAuditLog({
      actorId: adminId,
      action: `moderation.delete_${report.targetType}`,
      targetType: report.targetType,
      targetId: report.targetId,
      metadata: { reportId: report.id, reason: report.reason },
      ipHash,
    })

    return Response.json({ data: { status: 'deleted' } })
  }

  const status = action === 'resolve' ? 'resolved' : 'dismissed'
  const updated = await prisma.report.update({
    where: { id },
    data: {
      status,
      resolutionNote: note || null,
      reviewedById: adminId,
      reviewedAt,
    },
  })

  await writeAuditLog({
    actorId: adminId,
    action: `moderation.${action}_report`,
    targetType: report.targetType,
    targetId: report.targetId,
    metadata: { reportId: report.id, reason: report.reason },
    ipHash,
  })

  return Response.json({ data: { status: updated.status } })
}

import { auth } from '@/lib/auth/auth'
import { isAdmin } from '@/lib/admin'
import { prisma } from '@/lib/db'
import { jsonError } from '@/lib/community'
import { getLessonHref } from '@/lib/lesson-links'

export const dynamic = 'force-dynamic'

type ReportRow = Awaited<ReturnType<typeof loadReports>>[number]

async function loadReports(status: string) {
  return prisma.report.findMany({
    where: status === 'all' ? {} : { status },
    orderBy: { createdAt: 'desc' },
    take: 100,
    include: {
      reporter: { select: { id: true, name: true, image: true, email: true } },
      reviewedBy: { select: { id: true, name: true } },
    },
  })
}

export async function GET(request: Request) {
  const session = await auth()
  if (!isAdmin(session?.user?.email)) {
    return jsonError(403, 'FORBIDDEN', 'Maintainer access required.')
  }

  const url = new URL(request.url)
  const statusParam = url.searchParams.get('status')
  const status = ['open', 'resolved', 'dismissed', 'all'].includes(statusParam ?? '')
    ? (statusParam as string)
    : 'open'

  const reports = await loadReports(status)

  const projectIds = reports
    .filter((report) => report.targetType === 'project')
    .map((report) => report.targetId)
  const commentIds = reports
    .filter((report) => report.targetType === 'comment')
    .map((report) => report.targetId)
  const profileIds = reports
    .filter((report) => report.targetType === 'profile')
    .map((report) => report.targetId)

  const [projects, comments, profiles] = await Promise.all([
    projectIds.length
      ? prisma.project.findMany({
          where: { id: { in: projectIds } },
          select: {
            id: true,
            title: true,
            description: true,
            userId: true,
            user: { select: { name: true } },
          },
        })
      : [],
    commentIds.length
      ? prisma.comment.findMany({
          where: { id: { in: commentIds } },
          select: {
            id: true,
            body: true,
            userId: true,
            targetType: true,
            targetId: true,
            user: { select: { name: true } },
          },
        })
      : [],
    profileIds.length
      ? prisma.user.findMany({
          where: { id: { in: profileIds } },
          select: { id: true, name: true, email: true, image: true },
        })
      : [],
  ])

  const projectById = new Map(projects.map((project) => [project.id, project]))
  const commentById = new Map(comments.map((comment) => [comment.id, comment]))
  const profileById = new Map(profiles.map((profile) => [profile.id, profile]))

  const data = reports.map((report: ReportRow) => {
    let title = 'Deleted content'
    let preview = ''
    let href: string | null = null
    let authorName: string | null = null
    let available = true

    if (report.targetType === 'project') {
      const project = projectById.get(report.targetId)
      if (!project) {
        available = false
      } else {
        title = project.title
        preview = project.description.slice(0, 240)
        href = `/community/showcase/${project.id}`
        authorName = project.user.name
      }
    } else if (report.targetType === 'comment') {
      const comment = commentById.get(report.targetId)
      if (!comment) {
        available = false
      } else {
        title = 'Comment'
        preview = comment.body.slice(0, 240)
        href =
          comment.targetType === 'lesson'
            ? `${getLessonHref(comment.targetId)}#discussion-heading`
            : `/community/showcase/${comment.targetId}?comment=${comment.id}#discussion-heading`
        authorName = comment.user.name
      }
    } else {
      const profile = profileById.get(report.targetId)
      if (!profile) {
        available = false
      } else {
        title = profile.name ?? 'Unnamed learner'
        preview = profile.email
        href = `/community/people/${profile.id}`
        authorName = profile.name
      }
    }

    return {
      id: report.id,
      targetType: report.targetType,
      targetId: report.targetId,
      reason: report.reason,
      details: report.details,
      status: report.status,
      resolutionNote: report.resolutionNote,
      createdAt: report.createdAt.toISOString(),
      reviewedAt: report.reviewedAt?.toISOString() ?? null,
      reviewedBy: report.reviewedBy,
      reporter: {
        id: report.reporter.id,
        name: report.reporter.name,
        image: report.reporter.image,
      },
      target: { title, preview, href, authorName, available },
    }
  })

  return Response.json(
    { data, status },
    { headers: { 'Cache-Control': 'no-store' } }
  )
}

import { prisma } from '@/lib/db'
import { communityEnabled } from '@/lib/community'

type RouteContext = { params: Promise<{ id: string }> }

/** Public, immutable-by-URL avatar image served from PostgreSQL. */
export async function GET(_request: Request, { params }: RouteContext) {
  if (!communityEnabled()) {
    return new Response('Not found', { status: 404 })
  }

  const { id } = await params
  const avatar = await prisma.userAvatar.findUnique({
    where: { userId: id },
    select: { mime: true, data: true },
  })

  if (!avatar?.data) {
    return new Response('Not found', { status: 404 })
  }

  const binary = atob(avatar.data)
  const bytes = new Uint8Array(binary.length)
  for (let index = 0; index < binary.length; index += 1) {
    bytes[index] = binary.charCodeAt(index)
  }

  return new Response(bytes, {
    headers: {
      'Content-Type': avatar.mime,
      'Content-Length': String(bytes.byteLength),
      'Cache-Control': 'public, max-age=31536000, immutable',
      'X-Content-Type-Options': 'nosniff',
    },
  })
}

async function getComments(targetType: string, targetId: string) {

    where: {

      targetId,
      parentId: null,
    },
    orderBy: { createdAt: 'asc' },
    take: 200,
    include: {
      user: { select: publicUserSelect },
      replies: {
        orderBy: { createdAt: 'asc' },
        take: 50,

      },
    },
  })

  return comments.map((comment) => ({
    id: comment.id,
    body: comment.body,
    createdAt: comment.createdAt.toISOString(),
    author: comment.user,
    replies: comment.replies.map((reply) => ({
      id: reply.id,
      body: reply.body,
      createdAt: reply.createdAt.toISOString(),
      author: reply.user,
    })),
  }))
}

export async function GET(request: Request) {
  if (!communityEnabled()) {
    return jsonError(COMMUNITY_DISABLED.status, COMMUNITY_DISABLED.code, COMMUNITY_DISABLED.message)
  }

      },
    }
  )
}

export async function POST(request: Request) {
  if (!communityEnabled()) {
    return jsonError(COMMUNITY_DISABLED.status, COMMUNITY_DISABLED.code, COMMUNITY_DISABLED.message)
  }

  const session = await auth()
  const userId = session?.user?.id
  if (!userId) return jsonError(401, 'UNAUTHORIZED', 'Sign in to join the discussion.')

  const body = await request.json().catch(() => null)
  const parsed = commentSchema.safeParse(body)
  if (!parsed.success) {
    return Response.json(
      {
        error: {
          code: 'VALIDATION_ERROR',
          message: parsed.error.issues[0]?.message ?? 'Invalid comment',
        },
      },
      { status: 422 }
    )
  }

  const { targetType, targetId, body: commentBody, parentId } = parsed.data

  if (targetType === 'lesson' && !isKnownLesson(targetId)) {
    return jsonError(404, 'NOT_FOUND', 'That lesson does not exist.')
  }
  if (targetType === 'project') {
    const project = await prisma.project.findUnique({ where: { id: targetId }, select: { id: true } })
    if (!project) return jsonError(404, 'NOT_FOUND', 'That project does not exist.')
  }

  if (parentId) {
    const parent = await prisma.comment.findUnique({
      where: { id: parentId },
      select: { parentId: true, targetType: true, targetId: true },
    })
    if (!parent || parent.targetType !== targetType || parent.targetId !== targetId) {
      return jsonError(404, 'NOT_FOUND', 'The comment you are replying to does not exist.')
    }
    if (parent.parentId) {
      return jsonError(422, 'VALIDATION_ERROR', 'Replies can only be one level deep.')
    }
  }

  const recent = await prisma.comment.count({
    where: { userId, createdAt: { gte: sinceMinutes(10) } },
  })
  if (recent >= 12) {
    return jsonError(429, 'RATE_LIMITED', 'You are posting quickly — take a breath and try again in a few minutes.')
  }

  const comment = await prisma.comment.create({
    data: { userId, targetType, targetId, body: commentBody, parentId: parentId ?? null },
    include: { user: { select: publicUserSelect } },
  })

  after(async () => {
    try {
      if (parentId) {
        await notifyCommentReply(comment.id, parentId, userId)
      } else {
        await notifyCommentCreated(comment.id, targetType, targetId, userId)
      }
      await syncUserBadges(userId)
    } catch (error) {
      console.error('Failed to finish comment side effects', error)
    }
  })

  return Response.json(
    {
      data: {
        id: comment.id,
        body: comment.body,
        createdAt: comment.createdAt.toISOString(),
        author: comment.user,
        replies: [],
      },
    },
    { status: 201 }
  )
}

    const target = await prisma.user.findUnique({
      where: { id: targetUserId },
      select: { id: true },
    })
    if (!target) return jsonError(404, 'NOT_FOUND', 'Profile not found.')

    // Check if already following or requested
    const existing = await prisma.follow.findUnique({
      where: {
        followerId_followingId: {
          followerId: userId,
          followingId: targetUserId,
        },
      },
      select: { status: true },
    })

    if (existing) {
      if (existing.status === 'accepted') {
        return jsonError(422, 'VALIDATION_ERROR', 'You are already following this user.')
      }
      if (existing.status === 'pending') {
        return jsonError(422, 'VALIDATION_ERROR', 'You already have a pending follow request.')
      }
      // If rejected, allow new request
    }

    const follow = existing
      ? await prisma.follow.update({
          where: {
            followerId_followingId: {
              followerId: userId,
              followingId: targetUserId,
            },
          },
          data: { status: 'pending' },
        })
      : await prisma.follow.create({
          data: {
            followerId: userId,
            followingId: targetUserId,
            status: 'pending',
          },
        })

    try {
      await notifyFollowRequest(userId, targetUserId)
    } catch (error) {
      console.error('Failed to create follow request notification', error)
    }

    return Response.json({ data: { status: 'pending', requestId: follow.id } }, { status: 201 })
  }

  // Accept or reject an incoming follow request
  if (action !== 'accept' && action !== 'reject') {
    return jsonError(422, 'VALIDATION_ERROR', 'Choose accept, reject, or create.')
  }

  const { id } = await params
  const follow = await prisma.follow.findFirst({
    where: { id, followingId: userId, status: 'pending' },
    select: { id: true, followerId: true },
  })
  if (!follow) {
    return jsonError(404, 'NOT_FOUND', 'This follow request is no longer pending.')
  }

  const status = action === 'accept' ? 'accepted' : 'rejected'
  const updated = await prisma.follow.updateMany({
    where: { id, followingId: userId, status: 'pending' },
    data: { status },
  })
  if (updated.count === 0) {
    return jsonError(404, 'NOT_FOUND', 'This follow request is no longer pending.')
  }

  try {
    if (action === 'accept') {
      await notifyFollowAccepted(follow.followerId, userId)
    } else {
      await notifyFollowRejected(follow.followerId, userId)
    }
  } catch (error) {
    console.error('Failed to create follow response notification', error)
  }

  if (action === 'accept') {
    after(async () => {
      try {
        await syncUserBadges(userId)
      } catch (error) {
        console.error('Failed to sync follower badges', error)
      }
    })
  }

  return Response.json({
    data: { status: action === 'accept' ? 'following' : 'none' },
  })
}

  const userId = session?.user?.id
  if (!userId) return jsonError(401, 'UNAUTHORIZED', 'Sign in to follow someone.')

  const { id } = await params
  if (id === userId) return jsonError(400, 'INVALID_REQUEST', 'You cannot follow yourself.')

  const target = await prisma.user.findUnique({
    where: { id },
    select: { id: true },
  })
  if (!target) return jsonError(404, 'NOT_FOUND', 'Profile not found.')

  const [existing, reverse] = await Promise.all([
    prisma.follow.findUnique({
      where: {
        followerId_followingId: { followerId: userId, followingId: id },
      },
      select: { id: true, status: true },
    }),
    prisma.follow.findUnique({
      where: {
        followerId_followingId: { followerId: id, followingId: userId },
      },
      select: { id: true, status: true },
    }),
  ])

  if (existing?.status === 'accepted') {
    return jsonError(409, 'ALREADY_FOLLOWING', 'You already follow this learner.')
  }
  if (existing?.status === 'pending') {
    return jsonError(409, 'REQUEST_PENDING', 'Your follow request is already pending.')
  }
  if (reverse?.status === 'pending') {
    return jsonError(
      409,
      'INCOMING_REQUEST',
      'This learner has already sent you a follow request. Accept it from your profile.'
    )
  }

  const recent = await prisma.follow.count({
    where: { followerId: userId, createdAt: { gte: sinceMinutes(24 * 60) } },
  })
  if (recent >= 30) {
    return jsonError(429, 'RATE_LIMITED', 'You can send up to 30 follow requests per day.')
  }

  try {
    const follow = await prisma.follow.upsert({
      where: {
        followerId_followingId: { followerId: userId, followingId: id },
      },
      create: { followerId: userId, followingId: id, status: 'pending' },
      update: { status: 'pending' },
      select: { id: true },
    })

    try {
      await notifyFollowRequest(userId, id)
    } catch (error) {
      console.error('Failed to create follow request notification', error)
    }

    return Response.json({ data: { status: 'pending_outgoing', requestId: follow.id } })
  } catch (error) {
    if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === 'P2002') {
      return jsonError(409, 'REQUEST_PENDING', 'Your follow request is already pending.')
    }
    throw error
  }
}

export async function resolveUserId(identifier: string): Promise<string | null> {
  // First try exact ID match (could be CUID or shortened id like "admin")
  const byId = await prisma.user.findUnique({
    where: { id: identifier },
    select: { id: true },
  })
  if (byId) return byId.id

  // Then try username match (case-insensitive)
  if (identifier.length >= 3 && identifier.length <= 30) {
    const byUsername = await prisma.user.findUnique({
      where: { username: identifier.toLowerCase() },
      select: { id: true },
    })
    if (byUsername) return byUsername.id
  }

  return null
}

  'image/jpeg',
  'image/png',
  'image/webp',
  'image/gif',
])

/** Upload or replace the signed-in user's profile photo. */
export async function POST(request: Request) {
  if (!communityEnabled()) {
    return jsonError(COMMUNITY_DISABLED.status, COMMUNITY_DISABLED.code, COMMUNITY_DISABLED.message)
  }

  const session = await auth()
  const userId = session?.user?.id
  if (!userId) return jsonError(401, 'UNAUTHORIZED', 'Sign in first.')

  const contentLength = Number(request.headers.get('content-length') ?? 0)
  if (contentLength > MAX_BYTES + 1024 * 1024) {
    return jsonError(413, 'FILE_TOO_LARGE', 'Choose an image smaller than 3 MB.')
  }

  const form = await request.formData().catch(() => null)
  const file = form?.get('avatar')
  if (!(file instanceof File)) {
    return jsonError(422, 'VALIDATION_ERROR', 'Choose an image to upload.')
  }
  if (!ALLOWED_TYPES.has(file.type)) {
    return jsonError(422, 'INVALID_FILE_TYPE', 'Use a JPG, PNG, WebP, or GIF image.')
  }
  if (file.size === 0 || file.size > MAX_BYTES) {
    return jsonError(413, 'FILE_TOO_LARGE', 'Choose an image smaller than 3 MB.')
  }

  const data = Buffer.from(await file.arrayBuffer())
  const image = `/api/community/avatar/${userId}?v=${Date.now()}`

  await prisma.$transaction([
    prisma.userAvatar.upsert({
      where: { userId },
      create: { userId, mime: file.type, data: data.toString('base64') },
      update: { mime: file.type, data: data.toString('base64') },
    }),
    prisma.user.update({ where: { id: userId }, data: { image } }),
  ])

  return Response.json(
    { data: { image } },
    { headers: { 'Cache-Control': 'no-store' } }
  )
}

import { after } from 'next/server'
import { Prisma } from '@prisma/client'

import { syncUserBadges } from '@/lib/badge-awards'



type RouteContext = { params: Promise<{ id: string }> }

/**
 * Toggle an upvote. The unique (projectId, userId) constraint makes concurrent
 * double-clicks harmless; the denormalized counter moves in the same
 * transaction so sorting by popularity stays consistent with the vote rows.
 */
export async function POST(request: Request, { params }: RouteContext) {
  if (!communityEnabled()) {
    return jsonError(COMMUNITY_DISABLED.status, COMMUNITY_DISABLED.code, COMMUNITY_DISABLED.message)
  }

  const session = await auth()
  const userId = session?.user?.id
  if (!userId) return jsonError(401, 'UNAUTHORIZED', 'Sign in to upvote a project.')

  const { id } = await params
  const project = await prisma.project.findUnique({ where: { id }, select: { id: true } })
  if (!project) return jsonError(404, 'NOT_FOUND', 'Project not found.')

  const existing = await prisma.projectUpvote.findUnique({
    where: { projectId_userId: { projectId: id, userId } },
  })

  let upvoted: boolean
  if (existing) {
    await prisma.$transaction([
      prisma.projectUpvote.delete({ where: { id: existing.id } }),
      prisma.project.update({ where: { id }, data: { upvotes: { decrement: 1 } } }),
    ])
    upvoted = false
  } else {
    try {
      await prisma.$transaction([
        prisma.projectUpvote.create({ data: { projectId: id, userId } }),
        prisma.project.update({ where: { id }, data: { upvotes: { increment: 1 } } }),
      ])
    } catch (error) {
      // Lost a race with another click: the vote already exists, which is the
      // desired end state. Swallow only the unique-constraint error.
      if (!(error instanceof Prisma.PrismaClientKnownRequestError) || error.code !== 'P2002') {
        throw error
      }
    }
    upvoted = true
  }

  const updated = await prisma.project.findUnique({
    where: { id },
    select: { upvotes: true },
  })

  after(async () => {
    try {
      if (upvoted) await notifyProjectUpvote(id, userId)
      await syncUserBadges(userId)
    } catch (error) {
      console.error('Failed to finish upvote side effects', error)
    }
  })

  return Response.json({ data: { upvoted, upvotes: updated?.upvotes ?? 0 } })
}

    return jsonError(COMMUNITY_DISABLED.status, COMMUNITY_DISABLED.code, COMMUNITY_DISABLED.message)
  }

  const url = new URL(request.url)
  const track = url.searchParams.get('track')
  const tag = url.searchParams.get('tag')
  const sort = url.searchParams.get('sort') === 'popular' ? 'popular' : 'newest'
  const q = url.searchParams.get('q')?.trim()
  const limit = getLimit(url)
  const cursor = url.searchParams.get('cursor')

  const session = await auth()
  const userId = session?.user?.id ?? null

  const where: Prisma.ProjectWhereInput = {}
  if (track && tracks.some((entry) => entry.key === track)) where.track = track
  if (tag) where.tags = { has: tag }
  if (q) {
    where.OR = [
      { title: { contains: q, mode: 'insensitive' } },
      { description: { contains: q, mode: 'insensitive' } },
    ]
  }

  const projects = await prisma.project.findMany({
    where,
    take: limit + 1,
    ...(cursor ? { cursor: { id: cursor }, skip: 1 } : {}),
    orderBy:
      sort === 'popular'
        ? [{ upvotes: 'desc' }, { createdAt: 'desc' }]
        : [{ createdAt: 'desc' }],
    include: {
      user: { select: publicUserSelect },
      // A sentinel id keeps the response shape identical for signed-out visitors.
      upvoteRows: { where: { userId: userId ?? '__none__' }, select: { id: true } },
    },
  })

  const hasMore = projects.length > limit
  const page = hasMore ? projects.slice(0, limit) : projects
  const nextCursor = hasMore ? page[page.length - 1]?.id ?? null : null

  const ids = page.map((project) => project.id)
  const commentCounts = ids.length
    ? await prisma.comment.groupBy({
        by: ['targetId'],
        where: { targetType: 'project', targetId: { in: ids } },
        _count: { _all: true },
      })
    : []
  const countByProject = new Map(commentCounts.map((row) => [row.targetId, row._count._all]))

  return Response.json(
    {
      data: page.map((project) => serializeProject(project, countByProject.get(project.id) ?? 0)),
      nextCursor,
      sort,
    },
    {
      headers: {
        // Responses include the viewer's upvote state, so only the browser
        // may cache them. Filters and cursors produce distinct URLs.
        'Cache-Control': 'private, max-age=15, stale-while-revalidate=60',
      },
    }
  )

export const dynamic = 'force-dynamic'

const MAX_BYTES = 5 * 1024 * 1024
const ALLOWED_TYPES = new Set(['image/jpeg', 'image/png', 'image/webp', 'image/gif', 'image/avif'])

export async function POST(request: Request) {
  if (!communityEnabled()) {
    return jsonError(COMMUNITY_DISABLED.status, COMMUNITY_DISABLED.code, COMMUNITY_DISABLED.message)
  }

  const session = await auth()
  if (!session?.user?.id) {
    return jsonError(401, 'UNAUTHORIZED', 'Sign in to upload an image.')
  }

  if (!process.env.BLOB_READ_WRITE_TOKEN) {
    return jsonError(
      503,
      'UPLOADS_DISABLED',
      'Image uploads are not configured on this deployment. Paste a direct image URL instead.'
    )
  }

  const formData = await request.formData().catch(() => null)
  const file = formData?.get('file')

  if (!(file instanceof File)) {
    return jsonError(422, 'INVALID_FILE', 'Choose an image to upload.')
  }
  if (!ALLOWED_TYPES.has(file.type)) {
    return jsonError(422, 'INVALID_FILE', 'Use a JPEG, PNG, WebP, GIF, or AVIF image.')
  }
  if (file.size === 0 || file.size > MAX_BYTES) {
    return jsonError(422, 'INVALID_FILE', 'Images must be smaller than 5 MB.')
  }

  try {
    const extension = file.name.split('.').pop()?.toLowerCase() || 'img'
    const blob = await put(
      `community/${session.user.id}/${Date.now()}.${extension}`,
      file,
      {
        access: 'public',
        addRandomSuffix: true,
        contentType: file.type,
      }
    )

    return Response.json({ data: { url: blob.url } }, { status: 201 })
  } catch (error) {
    console.error('Project image upload failed', error)
    return jsonError(502, 'UPLOAD_FAILED', 'Could not upload that image. Try a smaller file.')
  }
}

export async function POST(request: Request) {
  if (!NEWSLETTER_ENABLED) {
    return jsonError(404, 'NEWSLETTER_DISABLED', 'The newsletter is not available.')
  }

  if (!databaseReady()) {
    return jsonError(503, 'NEWSLETTER_UNAVAILABLE', 'Newsletter signup is temporarily unavailable.')
  }
  if (!newsletterEmailConfigured()) {
    return jsonError(
      503,
      'NEWSLETTER_UNAVAILABLE',
      'Newsletter signup is temporarily unavailable. Please try again later.'
    )
  }

  const body = await request.json().catch(() => null)
  const parsed = subscribeSchema.safeParse(body)
  if (!parsed.success) {
    return Response.json(
      {
        error: {
          code: 'VALIDATION_ERROR',
          message: parsed.error.issues[0]?.message ?? 'Enter a valid email address.',
        },
      },
      { status: 422 }
    )
  }

  // Silently accept honeypot submissions so bots do not learn they were caught.
  if (parsed.data.website) {
    return Response.json({ data: { subscribed: true, confirmationRequired: false } })
  }

  const recent = await prisma.newsletterSubscriber.count({
    where: { createdAt: { gte: sinceMinutes(60) } },
  })
  if (recent >= 200) {
    return jsonError(429, 'RATE_LIMITED', 'Too many signups right now. Try again later.')
  }

  const subscriber = await prisma.newsletterSubscriber.upsert({
    where: { email: parsed.data.email },
    create: { email: parsed.data.email, source: parsed.data.source },
    update: {},
  })

  if (subscriber.confirmedAt) {
    return Response.json({
      data: {
        subscribed: true,
        confirmationRequired: false,
        alreadyConfirmed: true,
      },
    })
  }

  const sentRecently =
    subscriber.confirmationSentAt &&
    Date.now() - subscriber.confirmationSentAt.getTime() < 60_000

  if (sentRecently) {
    return Response.json({
      data: {
        subscribed: true,
        confirmationRequired: true,
        alreadySent: true,
      },
    })
  }

  try {
    await sendNewsletterConfirmation({
      email: subscriber.email,
      subscriberId: subscriber.id,
      origin: siteUrl,
    })
  } catch (error) {
    console.error(
      'Newsletter confirmation email failed:',
      error instanceof Error ? error.message : error
    )
    return jsonError(
      503,
      'NEWSLETTER_UNAVAILABLE',
      'We could not send the confirmation email. Please try again in a minute.'
    )
  }

  await prisma.newsletterSubscriber
    .update({
      where: { id: subscriber.id },
      data: { confirmationSentAt: new Date() },
    })
    .catch((error) => {
      console.error(
        'Could not record newsletter confirmation time:',
        error instanceof Error ? error.message : error
      )
    })

  return Response.json(
    {
      data: {
        subscribed: true,
        confirmationRequired: true,
        alreadySent: false,
      },
    },
    { status: 201 }
  )
}

import { after } from 'next/server'
import { z } from 'zod'
import { auth } from '@/lib/auth/auth'
import { prisma } from '@/lib/db'
import { syncUserBadges } from '@/lib/badge-awards'
import { syncUserCertificates } from '@/lib/certificates'
import { tracks } from '@/lib/curriculum'
import type { ProgressState } from '@/lib/progress-state'



const lessonProgressSchema = z.object({
  completed: z.boolean(),
  completedAt: z.string().max(40).nullish(),
  notes: z.string().max(10_000).nullish(),
})

const progressStateSchema = z.object({
  lessons: z
    .record(z.string().min(1).max(160), lessonProgressSchema)
    .refine((lessons) => Object.keys(lessons).length <= 500, {
      message: 'Too many lessons in payload',
    }),
  xp: z.number().int().min(0).max(10_000_000),
  streak: z.object({
    current: z.number().int().min(0).max(100_000),
    longest: z.number().int().min(0).max(100_000),
    lastActiveDate: z.string().max(10).nullable(),
  }),
  bookmarks: z.array(z.string().min(1).max(160)).max(500),
  lastVisited: z
    .object({
      track: z.string().max(160),
      module: z.string().max(160),
      lesson: z.string().max(160),
      timestamp: z.string().max(40),
    })
    .nullish(),
})

const xpByLesson = new Map<string, number>()
for (const track of tracks) {
  for (const mod of track.modules) {
    for (const lesson of mod.lessons) {
      xpByLesson.set(lesson.id, lesson.xpReward)
    }
  }
}

function toDate(value?: string | null): Date | null {
  if (!value) return null
  const date = new Date(value)
  return Number.isNaN(date.getTime()) ? null : date
}

const TRANSIENT_DB_CODES = new Set(['P1001', 'P1017', 'P2002', 'P2028', 'P2034'])

function isTransientDbError(error: unknown): boolean {
  return Boolean(
    error &&
      typeof error === 'object' &&
      'code' in error &&
      typeof (error as { code?: unknown }).code === 'string' &&
      TRANSIENT_DB_CODES.has((error as { code: string }).code)
  )
}

async function replaceProgress(
  userId: string,
  entries: [string, z.infer<typeof lessonProgressSchema>][],
  statsData: {
    xp: number
    streakCurrent: number
    streakLongest: number
    streakLastActive: Date | null
    bookmarks: string[]
    lastVisitedTrack: string | null
    lastVisitedModule: string | null
    lastVisitedLesson: string | null
    lastVisitedAt: Date | null
  }
) {
  // Two tabs can save at the same time. A transaction-scoped advisory lock
  // serializes replacement per learner, so delete + bulk insert cannot race
  // into a unique-constraint failure on (userId, lessonKey).
  for (let attempt = 0; attempt < 3; attempt += 1) {
    try {
      await prisma.$transaction(
        async (tx) => {
          await tx.$queryRaw`SELECT pg_advisory_xact_lock(hashtext(${userId}))::text AS lock`

          await tx.userProgress.deleteMany({ where: { userId } })

          if (entries.length > 0) {
            await tx.userProgress.createMany({
              data: entries.map(([lessonKey, progress]) => ({
                userId,
                lessonKey,
                completed: progress.completed,
                notes: progress.notes?.trim() ? progress.notes : null,
                xpEarned: progress.completed ? (xpByLesson.get(lessonKey) ?? 0) : 0,
                completedAt: progress.completed
                  ? toDate(progress.completedAt) ?? new Date()
                  : null,
              })),
            })
          }

          await tx.userStats.upsert({
            where: { userId },
            create: { userId, ...statsData },
            update: statsData,
          })
        },
        { maxWait: 10_000, timeout: 30_000 }
      )
      return
    } catch (error) {
      if (attempt < 2 && isTransientDbError(error)) {
        await new Promise((resolve) => setTimeout(resolve, 150 * (attempt + 1)))
        continue
      }
      throw error
    }
  }
}

export async function GET() {

  const userId = session?.user?.id ?? null

export async function PUT(request: Request) {
  const session = await auth()
  const userId = session?.user?.id ?? null
  if (!userId) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const body = await request.json().catch(() => null)
  const parsed = progressStateSchema.safeParse(
    body && typeof body === 'object' && 'state' in body ? body.state : null
  )
  if (!parsed.success) {
    return Response.json({ error: 'Invalid progress payload' }, { status: 400 })
  }

  const payload = parsed.data

  // Only persist lessons that have been completed or annotated.
  const entries = Object.entries(payload.lessons).filter(
    ([, progress]) => progress.completed || Boolean(progress.notes?.trim())
  )
  // Never trust a client-provided XP total: recompute it from the curriculum.
  const xp = entries.reduce(
    (sum, [key, progress]) =>
      progress.completed ? sum + (xpByLesson.get(key) ?? 0) : sum,
    0
  )

  const lastVisited = payload.lastVisited
    ? {
        lastVisitedTrack: payload.lastVisited.track,
        lastVisitedModule: payload.lastVisited.module,
        lastVisitedLesson: payload.lastVisited.lesson,
        lastVisitedAt: toDate(payload.lastVisited.timestamp),
      }
    : {
        lastVisitedTrack: null,
        lastVisitedModule: null,
        lastVisitedLesson: null,
        lastVisitedAt: null,
      }

  const streakLastActive = toDate(payload.streak.lastActiveDate)

  const statsData = {
    xp,
    streakCurrent: payload.streak.current,
    streakLongest: payload.streak.longest,
    streakLastActive,
    bookmarks: Array.from(new Set(payload.bookmarks)),
    ...lastVisited,
  }

  await replaceProgress(userId, entries, statsData)

  const sanitizedLessons: ProgressState['lessons'] = {}
  for (const [lessonKey, progress] of entries) {
    sanitizedLessons[lessonKey] = {
      completed: progress.completed,
      ...(progress.completed
        ? { completedAt: toDate(progress.completedAt)?.toISOString() ?? new Date().toISOString() }
        : {}),
      ...(progress.notes?.trim() ? { notes: progress.notes } : {}),
    }
  }

  const state: ProgressState = {
    lessons: sanitizedLessons,
    xp,
    streak: payload.streak,
    bookmarks: Array.from(new Set(payload.bookmarks)),
    lastVisited: payload.lastVisited
      ? { ...payload.lastVisited, timestamp: toDate(payload.lastVisited.timestamp)?.toISOString() ?? new Date().toISOString() }
      : null,
  }

  // Persist earned badges after the response has been sent so saving progress
  // stays fast. The client toast still unlocks instantly from local state.
  after(async () => {
    try {
      await Promise.all([syncUserBadges(userId), syncUserCertificates(userId)])
    } catch (error) {
      console.error('Failed to sync badges or certificates after progress update', error)
    }
  })

  return Response.json({ state })
}
