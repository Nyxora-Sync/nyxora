const typeOrder: Record<SearchResultType, number> = {
  lesson: 0,
  module: 1,
  track: 2,
  page: 3,
  faq: 4,
  resource: 5,
  project: 6,
}

let cachedStaticEntries: SearchResult[] | null = null

function getStaticEntries(): SearchResult[] {
  if (cachedStaticEntries) return cachedStaticEntries

  const entries = new Map<string, SearchResult>()

  for (const page of staticPages) {
    entries.set(page.href, page)
  }

  for (const track of tracks) {
    entries.set(`/learn/${track.key}`, {
      id: `track-${track.key}`,
      type: 'track',
      title: track.name,
      description: track.description,
      href: `/learn/${track.key}`,
      meta: `${track.modules.length} modules`,
      keywords: `${track.key} track modules curriculum`,
    })

    for (const mod of track.modules) {
      const moduleHref = canonicalLearnPath(track.key, mod.key)
      if (!entries.has(moduleHref)) {
        entries.set(moduleHref, {
          id: `module-${mod.key}`,
          type: 'module',
          title: mod.name,
          description: mod.description,
          href: moduleHref,
          meta: `${mod.lessons.length} lessons · about ${mod.estimatedHours} hours`,
          keywords: `${track.name} ${mod.key} ${mod.lessons.map((lesson) => lesson.title).join(' ')}`,
        })
      }

      for (const lesson of mod.lessons) {
        const lessonHref = canonicalLearnPath(track.key, mod.key, lesson.id)
        if (!entries.has(lessonHref)) {
          entries.set(lessonHref, {
            id: `lesson-${lesson.id}`,
            type: 'lesson',
            title: lesson.title,
            description: lesson.description,
            href: lessonHref,
            meta: `${mod.name} · ${lesson.estimatedMinutes} min · ${lesson.difficulty}`,
            keywords: [
              track.name,
              mod.name,
              lesson.difficulty,
              ...lesson.keyTakeaways,
              ...lesson.resources.map((resource) => resource.title),
            ].join(' '),
          })
        }
      }
    }
  }

  for (const resource of resourcesData) {
    entries.set(resource.url, {
      id: `resource-${resource.id}`,
      type: 'resource',
      title: resource.title,
      description: resource.description,
      href: resource.url,
      external: true,
      meta: `${resource.provider} · ${resource.type} · ${resource.level}`,
      keywords: `${resource.topic} ${resource.tags.join(' ')} ${resource.provider}`,
    })
  }

  for (const section of faqSections) {
    for (const item of section.items) {
      entries.set(`/faq#${item.slug}`, {
        id: `faq-${item.slug}`,
        type: 'faq',
        title: item.question,
        description: item.answer,
        href: `/faq#${item.slug}`,
        meta: `FAQ · ${section.title}`,
        keywords: `${section.title} ${section.description} ${item.answer}`,
      })
    }
  }

  cachedStaticEntries = [...entries.values()]
  return cachedStaticEntries
}

function normalize(value: string): string {
  return value.toLowerCase().replace(/\s+/g, ' ').trim()
}

function queryTerms(query: string): string[] {
  return normalize(query).split(' ').filter(Boolean)
}

function scoreEntry(entry: SearchResult, normalizedQuery: string, terms: string[]): number {
  const title = normalize(entry.title)
  const meta = normalize(entry.meta ?? '')
  const keywords = normalize(entry.keywords ?? '')
  const haystack = `${title} ${normalize(entry.description)} ${meta} ${keywords}`

  if (!terms.every((term) => haystack.includes(term))) return -1

  let score = typeOrder[entry.type] * 0.1

  if (title === normalizedQuery) score += 100
  else if (title.startsWith(normalizedQuery)) score += 60
  else if (title.includes(normalizedQuery)) score += 30

  for (const term of terms) {
    if (title.includes(term)) score += 12
    if (meta.includes(term)) score += 4
    if (keywords.includes(term)) score += 2
  }

  return score
}

function sortResults(results: SearchResult[], normalizedQuery: string, terms: string[]): SearchResult[] {
  return results
    .map((entry) => ({ entry, score: scoreEntry(entry, normalizedQuery, terms) }))
    .filter((item) => item.score >= 0)
    .sort((a, b) => b.score - a.score || a.entry.title.localeCompare(b.entry.title))
    .map((item) => item.entry)
}

async function searchProjects(terms: string[]): Promise<SearchResult[]> {
  if (!communityEnabled() || terms.length === 0) return []

  try {
    const projects = await prisma.project.findMany({
      where: {
        OR: terms.map((term) => ({
          OR: [
            { title: { contains: term, mode: 'insensitive' } },
            { description: { contains: term, mode: 'insensitive' } },
            { tags: { has: term } },
            { modules: { has: term } },
          ],
        })),
      },
      orderBy: [{ upvotes: 'desc' }, { createdAt: 'desc' }],
      take: 8,
      select: {
        id: true,
        title: true,
        description: true,
        tags: true,
        track: true,
        upvotes: true,
      },
    })

    return projects.map((project) => ({
      id: `project-${project.id}`,
      type: 'project' as const,
      title: project.title,
      description: project.description,
      href: `/community/showcase/${project.id}`,
      meta: `${project.track} track · ${project.upvotes} ${project.upvotes === 1 ? 'upvote' : 'upvotes'}`,
      keywords: project.tags.join(' '),
    }))
  } catch {
    // Search should still work when the community database is unavailable.
    return []
  }
}

function withTimeout<T>(promise: Promise<T>, ms: number, fallback: T): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined

  const timeout = new Promise<T>((resolve) => {
    timer = setTimeout(() => resolve(fallback), ms)
  })

  return Promise.race([promise, timeout]).finally(() => {
    if (timer) clearTimeout(timer)
  })
}

export async function searchSite(query: string): Promise<SearchResult[]> {
  const cleanedQuery = query.trim().slice(0, 120)
  const normalizedQuery = normalize(cleanedQuery)
  if (!normalizedQuery) return []

  const terms = queryTerms(cleanedQuery).slice(0, 8)
  const staticResults = sortResults(getStaticEntries(), normalizedQuery, terms)
  // Static results should never wait on a slow or unavailable community database.
  const projectResults = sortResults(
    await withTimeout<SearchResult[]>(searchProjects(terms), 1500, []),
    normalizedQuery,
    terms
  )

  return [...staticResults, ...projectResults]
}

export function groupSearchResults(results: SearchResult[]): Array<{
  type: SearchResultType
  label: string
  results: SearchResult[]
}> {
  const labels: Record<SearchResultType, string> = {
    lesson: 'Lessons',
    module: 'Modules',
    track: 'Tracks',
    page: 'Pages',
    faq: 'FAQ',
    resource: 'Resources',
    project: 'Showcase projects',
  }

  const grouped = new Map<SearchResultType, SearchResult[]>()
  for (const result of results) {
    const list = grouped.get(result.type) ?? []
    list.push(result)
    grouped.set(result.type, list)
  }

  return ([...grouped.entries()] as Array<[SearchResultType, SearchResult[]]>)
    .sort(([a], [b]) => typeOrder[a] - typeOrder[b])
    .map(([type, items]) => ({ type, label: labels[type], results: items }))
}

import { getRequestIp, hashIp } from '@/lib/security'

describe('request security helpers', () => {
  it('takes the first forwarded address', () => {
    const request = new Request('https://example.com', {
      headers: { 'x-forwarded-for': '203.0.113.9, 10.0.0.1' },
    })
    expect(getRequestIp(request)).toBe('203.0.113.9')
  })

  it('falls back to Cloudflare and real-IP headers', () => {
    expect(
      getRequestIp(
        new Request('https://example.com', {
          headers: { 'cf-connecting-ip': '198.51.100.4' },
        })
      )
    ).toBe('198.51.100.4')
  })

  it('hashes IPs deterministically without keeping the raw value', () => {
    const first = hashIp('203.0.113.9')
    const second = hashIp('203.0.113.9')
    expect(first).toBe(second)
    expect(first).not.toContain('203.0.113.9')
    expect(first).toHaveLength(64)
  })
})

import { createHash } from 'node:crypto'
import { prisma } from '@/lib/db'

const LOGIN_WINDOW_MS = 15 * 60 * 1000
const MAX_FAILED_LOGINS = 5

/** Best-effort client IP from the proxy headers Vercel and Cloudflare set. */
export function getRequestIp(request: Request): string {
  const forwarded = request.headers.get('x-forwarded-for')
  if (forwarded) return forwarded.split(',')[0]!.trim()

  return (
    request.headers.get('cf-connecting-ip') ??
    request.headers.get('x-real-ip') ??
    'unknown'
  )
}

/**
 * Pseudonymize an IP before storing it. We never persist raw addresses in the
 * audit log, but repeated abuse can still be correlated.
 */
export function hashIp(ip: string): string {
  const salt = process.env.AUTH_SECRET || 'nyxora-ip-hash'
  return createHash('sha256').update(`${salt}:${ip}`).digest('hex')
}

export async function isAdminLoginRateLimited(ipHash: string): Promise<boolean> {
  const failures = await prisma.adminAuditLog.count({
    where: {
      action: 'admin.login_failed',
      ipHash,
      createdAt: { gte: new Date(Date.now() - LOGIN_WINDOW_MS) },
    },
  })

  return failures >= MAX_FAILED_LOGINS
}

export async function recordAdminLogin({
  actorId,
  ipHash,
  success,
}: {
  actorId?: string | null
  ipHash: string
  success: boolean
}): Promise<void> {
  try {
    await prisma.adminAuditLog.create({
      data: {
        actorId: success ? actorId ?? null : null,
        action: success ? 'admin.login_success' : 'admin.login_failed',
        ipHash,
      },
    })
  } catch (error) {
    // Audit logging must never turn a valid login into an outage.
    console.error('Failed to record admin login audit event', error)
  }
}

export async function writeAuditLog({
  actorId,
  action,
  targetType,
  targetId,
  metadata,
  ipHash,
}: {
  actorId: string
  action: string
  targetType?: string
  targetId?: string
  metadata?: Record<string, unknown>
  ipHash?: string
}): Promise<void> {
  await prisma.adminAuditLog.create({
    data: {
      actorId,
      action,
      targetType,
      targetId,
      metadata: metadata as object | undefined,
      ipHash,
    },
  })
}

  trackKey: string,

  lessonId?: string

export interface ProfileViewCounts {
  allTime: number
  month: number
  week: number
  day: number
}

function startOfUtcDay(date = new Date()): Date {
  return new Date(
    Date.UTC(date.getUTCFullYear(), date.getUTCMonth(), date.getUTCDate())
  )
}

/**
 * Visitor counts are for unique people, not page loads. ProfileView stores at
 * most one row per viewer per UTC day, and these queries count distinct
 * viewer keys across the requested window.
 */
export async function getProfileViewCounts(
  profileUserId: string
): Promise<ProfileViewCounts> {
  const now = new Date()
  const dayStart = new Date(now.getTime() - 24 * 60 * 60 * 1000)
  const weekStart = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)
  const monthStart = new Date(
    Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), 1)
  )

  const rows = await prisma.$queryRaw<
    Array<{ allTime: number; month: number; week: number; day: number }>
  >`
    SELECT
      COUNT(DISTINCT "viewerKey")::int AS "allTime",
      COUNT(DISTINCT "viewerKey") FILTER (WHERE "viewedAt" >= ${monthStart})::int AS "month",
      COUNT(DISTINCT "viewerKey") FILTER (WHERE "viewedAt" >= ${weekStart})::int AS "week",
      COUNT(DISTINCT "viewerKey") FILTER (WHERE "viewedAt" >= ${dayStart})::int AS "day"
    FROM "ProfileView"
    WHERE "profileUserId" = ${profileUserId}
  `

  return rows[0] ?? { allTime: 0, month: 0, week: 0, day: 0 }
}

async function anonymousViewerKey(request: Request): Promise<string> {
  const ip =
    request.headers.get('cf-connecting-ip') ??
    request.headers.get('x-forwarded-for')?.split(',')[0]?.trim() ??
    'unknown'
  const userAgent = request.headers.get('user-agent') ?? ''
  const secret = process.env.AUTH_SECRET ?? 'nyxora-profile-view'
  const input = new TextEncoder().encode(`${ip}\n${userAgent}\n${secret}`)
  const digest = await globalThis.crypto.subtle.digest('SHA-256', input)
  const hash = Array.from(new Uint8Array(digest))
    .slice(0, 16)
    .map((byte) => byte.toString(16).padStart(2, '0'))
    .join('')
  return `anon:${hash}`
}

/**
 * Records one visit per visitor per day. Self-views are deliberately ignored.
 * A failure here must never prevent someone from viewing a profile.
 */
export async function recordProfileView(
  request: Request,
  profileUserId: string,
  viewerId: string | null
): Promise<void> {
  if (viewerId === profileUserId) return

  try {
    const viewerKey = viewerId
      ? `user:${viewerId}`
      : await anonymousViewerKey(request)
    const now = new Date()
    const viewDate = startOfUtcDay(now)

    await prisma.profileView.upsert({
      where: {
        profileUserId_viewerKey_viewDate: {
          profileUserId,
          viewerKey,
          viewDate,
        },
      },
      create: {
        profileUserId,
        viewerId,
        viewerKey,
        viewDate,
      },
      update: { viewedAt: now },
    })
  } catch {
    // Visitor analytics are best-effort.
  }
}

export async function getFollowRelationship(
  viewerId: string | null,
  profileUserId: string
): Promise<FollowRelationship> {
  if (!viewerId || viewerId === profileUserId) {
    return {
      status: viewerId === profileUserId ? 'self' : 'none',
      requestId: null,
    }
  }

  const rows = await prisma.follow.findMany({
    where: {
      OR: [
        { followerId: viewerId, followingId: profileUserId },
        { followerId: profileUserId, followingId: viewerId },
      ],
    },
    select: {
      id: true,
      followerId: true,
      followingId: true,
      status: true,
    },
  })

  const outgoing = rows.find((row) => row.followerId === viewerId)
  const incoming = rows.find((row) => row.followerId === profileUserId)

  if (outgoing?.status === 'accepted') {
    return { status: 'following', requestId: null }
  }
  if (outgoing?.status === 'pending') {
    return { status: 'pending_outgoing', requestId: null }
  }
  if (incoming?.status === 'pending') {
    return { status: 'pending_incoming', requestId: incoming.id }
  }
  return { status: 'none', requestId: null }
}

export async function getFollowCounts(userId: string) {
  const rows = await prisma.$queryRaw<
    Array<{ followers: number; following: number }>
  >`
    SELECT
      COUNT(*) FILTER (WHERE "followingId" = ${userId} AND status = 'accepted')::int AS "followers",
      COUNT(*) FILTER (WHERE "followerId" = ${userId} AND status = 'accepted')::int AS "following"
    FROM "Follow"
  `
  return rows[0] ?? { followers: 0, following: 0 }
}

import { formatDistanceToNow } from 'date-fns'

export function timeAgo(iso: string): string {
  const date = new Date(iso)
  if (Number.isNaN(date.getTime())) return ''
  return formatDistanceToNow(date, { addSuffix: true })
}

export function shortDate(iso: string): string {
  const date = new Date(iso)
  if (Number.isNaN(date.getTime())) return ''
  return date.toLocaleDateString(undefined, { month: 'short', year: 'numeric' })
}

export function formatDuration(minutes: number): string {
  if (minutes < 60) return `${minutes} min`
  const hours = Math.floor(minutes / 60)
  const mins = minutes % 60
  return mins > 0 ? `${hours}h ${mins}m` : `${hours}h`
}

export function calculateLevel(xp: number): { level: number; xpForNext: number; progress: number } {
  const level = Math.floor(Math.sqrt(xp / 100)) + 1
  const xpForCurrent = (level - 1) ** 2 * 100
  const xpForNext = level ** 2 * 100
  const progress = ((xp - xpForCurrent) / (xpForNext - xpForCurrent)) * 100
  return { level, xpForNext, progress: Math.min(100, Math.max(0, progress)) }
}

import { auth } from '@/lib/auth/auth'
import { isAdmin } from '@/lib/admin'
import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

export async function adminMiddleware(request: NextRequest) {
  const session = await auth()
  
  if (!session?.user?.email || !isAdmin(session.user.email)) {
    return NextResponse.redirect(new URL('/auth/signin?error=Unauthorized', request.url))
  }

  return NextResponse.next()
}

export const config = {
  matcher: ['/admin/:path*'],
}
